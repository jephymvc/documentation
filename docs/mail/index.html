Mail

Overview

{{site.title}} provides a powerful Mailer class built on top of PHPMailer, offering a clean, intuitive interface for sending emails with support for SMTP, templates, attachments, and hooks. The Mailer integrates seamlessly with Smarty templates and the framework's hook system.

Reusable Email Service

php
// jephy-mvc/app/services/EmailService.php
namespace App\Services;

use App\Core\Mailer;
use App\Core\Config;

class EmailService
{
    private $mailer;
    private $config;
    
    public function __construct()
    {
        $this->mailer = new Mailer();
        $this->config = Config::getInstance();
    }
    
    public function sendWelcomeEmail($user)
    {
        $body = $this->mailer->template('emails.welcome', [
            'name' => $user->name,
            'email' => $user->email,
            'verificationUrl' => $this->generateVerificationUrl($user),
            'siteName' => $this->config->get('app.name')
        ]);
        
        return $this->mailer->send($user->email, 'Welcome to ' . $this->config->get('app.name'), $body);
    }
    
    public function sendPasswordResetEmail($user, $token)
    {
        $body = $this->mailer->template('emails.password-reset', [
            'name' => $user->name,
            'resetUrl' => $this->config->get('app.url') . '/reset-password?token=' . $token,
            'expiresIn' => '60 minutes'
        ]);
        
        return $this->mailer->send($user->email, 'Reset Your Password', $body);
    }
    
    public function sendOrderConfirmation($order, $user)
    {
        $items = [];
        foreach ($order->items as $item) {
            $items[] = [
                'name' => $item->product->name,
                'quantity' => $item->quantity,
                'price' => $item->price,
                'subtotal' => $item->quantity * $item->price
            ];
        }
        
        $body = $this->mailer->template('emails.order-confirmation', [
            'orderId' => $order->id,
            'customerName' => $user->name,
            'items' => $items,
            'total' => $order->total,
            'shippingAddress' => $order->shipping_address,
            'estimatedDelivery' => date('F j, Y', strtotime('+5 days'))
        ]);
        
        return $this->mailer->send($user->email, "Order Confirmation #{$order->id}", $body);
    }
    
    public function sendNewsletter($subscriber, $content)
    {
        $body = $this->mailer->template('emails.newsletter', [
            'subscriberName' => $subscriber->name,
            'content' => $content,
            'unsubscribeUrl' => $this->config->get('app.url') . '/unsubscribe?token=' . $subscriber->token
        ]);
        
        return $this->mailer->send($subscriber->email, 'Your Weekly Newsletter', $body);
    }
    
    private function generateVerificationUrl($user)
    {
        $token = bin2hex(random_bytes(32));
        // Store token in database
        return $this->config->get('app.url') . '/verify-email?token=' . $token . '&email=' . urlencode($user->email);
    }
}

Bulk Email Sending

Sending to Multiple Recipients Efficiently

php
class BulkMailService
{
    private $mailer;
    private $batchSize = 100;
    
    public function __construct()
    {
        $this->mailer = new Mailer();
    }
    
    public function sendNewsletterToSubscribers($subscribers, $subject, $template, $templateData)
    {
        $successCount = 0;
        $failCount = 0;
        $chunks = array_chunk($subscribers, $this->batchSize);
        
        foreach ($chunks as $chunk) {
            foreach ($chunk as $subscriber) {
                try {
                    // Personalize template for each subscriber
                    $personalizedData = array_merge($templateData, [
                        'name' => $subscriber->name,
                        'email' => $subscriber->email
                    ]);
                    
                    $body = $this->mailer->template($template, $personalizedData);
                    
                    if ($this->mailer->send($subscriber->email, $subject, $body)) {
                        $successCount++;
                    } else {
                        $failCount++;
                    }
                    
                    // Small delay to avoid rate limiting
                    usleep(50000); // 50ms
                    
                } catch (\Exception $e) {
                    $failCount++;
                    error_log("Failed to send to {$subscriber->email}: " . $e->getMessage());
                }
            }
            
            // Delay between batches
            sleep(1);
        }
        
        return [
            'success' => $successCount,
            'failed' => $failCount,
            'total' => count($subscribers)
        ];
    }
}

Queue-Based Email Sending

Email Job Class

php
// jephy-mvc/app/jobs/SendEmailJob.php
namespace App\Jobs;

use Core\Queue\Job;
use App\Core\Mailer;

class SendEmailJob extends Job
{
    protected $to;
    protected $subject;
    protected $body;
    protected $attachments;
    protected $options;
    
    public function __construct($to, $subject, $body, $attachments = [], $options = [])
    {
        $this->to = $to;
        $this->subject = $subject;
        $this->body = $body;
        $this->attachments = $attachments;
        $this->options = $options;
    }
    
    public function handle(Mailer $mailer)
    {
        // Set custom from if provided
        if (isset($this->options['from'])) {
            $mailer->setFrom($this->options['from']['email'], $this->options['from']['name'] ?? '');
        }
        
        // Add CC if provided
        if (isset($this->options['cc'])) {
            foreach ($this->options['cc'] as $cc) {
                $mailer->addCC($cc);
            }
        }
        
        // Add BCC if provided
        if (isset($this->options['bcc'])) {
            foreach ($this->options['bcc'] as $bcc) {
                $mailer->addBCC($bcc);
            }
        }
        
        // Set HTML format
        if (isset($this->options['is_html'])) {
            $mailer->isHTML($this->options['is_html']);
        }
        
        return $mailer->send($this->to, $this->subject, $this->body, $this->attachments);
    }
    
    public function failed(\Exception $e)
    {
        error_log("Failed to send email to {$this->to}: " . $e->getMessage());
        
        // Store failed email for retry
        FailedEmail::create([
            'to' => is_array($this->to) ? json_encode($this->to) : $this->to,
            'subject' => $this->subject,
            'error' => $e->getMessage(),
            'attempts' => $this->attempts
        ]);
    }
}

// Usage - Queue email for background sending
dispatch(new SendEmailJob($user->email, 'Welcome!', $body))->delay(0);

// Queue with delay (send after 5 minutes)
dispatch(new SendEmailJob($user->email, 'Reminder', $reminderBody))->delay(300);

Rate Limiting for Bulk Sends

php
class RateLimitedMailer extends Mailer
{
    private $rateLimit = 10; // emails per minute
    private $lastSendTimes = [];
    
    public function send($to, $subject, $body, $attachments = [])
    {
        $this->enforceRateLimit();
        return parent::send($to, $subject, $body, $attachments);
    }
    
    private function enforceRateLimit()
    {
        $now = microtime(true);
        $this->lastSendTimes = array_filter($this->lastSendTimes, function($time) use ($now) {
            return ($now - $time) < 60; // Keep only last minute
        });
        
        if (count($this->lastSendTimes) >= $this->rateLimit) {
            $sleep = 60 - ($now - min($this->lastSendTimes));
            if ($sleep > 0) {
                sleep((int)ceil($sleep));
            }
        }
        
        $this->lastSendTimes[] = microtime(true);
    }
}

3. Validate Email Addresses

php
// BAD - Send without validation

if (filter_var($email, FILTER_VALIDATE_EMAIL)) {

}

4. Handle Errors Gracefully

php

// GOOD - Handle and log errors
try {
    $result = $mailer->send($email, 'Subject', 'Body');
    if (!$result) {
        error_log("Failed to send email to: $email");
    }
} catch (\Exception $e) {
    error_log("Email error: " . $e->getMessage());
    // Fallback to alternative method
}

5. Rate Limit Bulk Sends

php
// BAD - Send all at once
foreach ($subscribers as $subscriber) {
    $mailer->send($subscriber->email, 'Newsletter', $body);
}

// GOOD - Rate limit
$delay = 0;
foreach ($subscribers as $subscriber) {
    dispatch(new SendEmailJob($subscriber->email, 'Newsletter', $body))->delay($delay);
    $delay += 1; // 1 second between sends
}

Error Handling

php
class SecureMailer extends Mailer
{
    public function sendSecure($to, $subject, $body, $attachments = [])
    {
        try {
            // Validate email addresses
            $validTo = is_array($to) ? array_filter($to, 'filter_var', FILTER_VALIDATE_EMAIL) : $to;
            
            if (empty($validTo)) {
                throw new \Exception("No valid recipient addresses");
            }
            
            // Sanitize subject
            $subject = strip_tags($subject);
            
            // Send email
            $result = $this->send($validTo, $subject, $body, $attachments);
            
            if (!$result) {
                throw new \Exception("Mailer returned false");
            }
            
            return ['success' => true, 'message' => 'Email sent'];
            
        } catch (\PHPMailer\PHPMailer\Exception $e) {
            error_log("PHPMailer Error: " . $e->getMessage());
            return ['success' => false, 'error' => 'Mailer error occurred'];
            
        } catch (\Exception $e) {
            error_log("General Error: " . $e->getMessage());
            return ['success' => false, 'error' => 'Failed to send email'];
        }
    }
}

php

// jephy-mvc/app/controllers/ContactController.php
namespace App\Controllers;

use App\Core\Controller;
use App\Services\EmailService;

class ContactController extends Controller
{
    private $emailService;
    
    public function __construct(EmailService $emailService)
    {
        $this->emailService = $emailService;
    }
    
    public function submit()
    {
        $data = $this->request->post();
        
        // Validate form
        $validator = new Validation();
        $rules = [
            'name' => 'required|min:2',
            'email' => 'required|email',
            'message' => 'required|min:10'
        ];
        
        if (!$validator->validate($data, $rules)) {
            return $this->json(['errors' => $validator->getErrors()], 422);
        }
        
        // Prepare email content
        $body = $this->view->render('emails.contact', [
            'name' => $data['name'],
            'email' => $data['email'],
            'message' => $data['message'],
            'ip' => $_SERVER['REMOTE_ADDR']
        ]);
        
        // Send to admin
        $mailer = new Mailer();
        $mailer->setFrom($data['email'], $data['name']);
        $mailer->addReplyTo($data['email'], $data['name']);
        
        $result = $mailer->send('admin@example.com', 'Contact Form: ' . $data['subject'] ?? 'No Subject', $body);
        
        if ($result) {
            // Send auto-reply to user
            $autoReplyBody = $this->view->render('emails.contact-auto-reply', [
                'name' => $data['name']
            ]);
            $mailer->send($data['email'], 'Thank you for contacting us', $autoReplyBody);
            
            return $this->json(['success' => true, 'message' => 'Message sent successfully']);
        }
        
        return $this->json(['success' => false, 'message' => 'Failed to send message'], 500);
    }
}
