Database Overview

{{site.title}} supports RDMS CRUD transactions with the QueryBuilder and Model core classes.

Query Builder provides a fluent, intuitive interface for building database queries programmatically. It supports method chaining, relationships, eager loading, batch operations, and raw expressions.

ORM provides an elegant Active Record implementation with support for relationships, query building, eager loading, and event hooks. Each database table has a corresponding Model class that interacts with that table.



Getting Started with QueryBuilder

Basic Usage:

php

use App\Core\QueryBuilder;

// Create a new query builder instance
$query = new QueryBuilder('users');

// Or use with entity class for ORM-style
$query = (new QueryBuilder('users'))->setEntity(User::class);

QueryBuilder Select Queries

Basic Select

php
$db = new QueryBuilder('users');

// Select all users
$users = $db->get();

// Select specific columns
$users = $db->select('id', 'name', 'email')->get();

// Select with array
$users = $db->select(['id', 'name', 'email'])->get();

// Select first record
$user = $db->first();

// Find by ID
$user = $db->find(1);

Where Clauses

php

$db = new QueryBuilder('users');

// Basic where
$users = $db->where('status', 'active')->get();

// Where with operator
$users = $db->where('age', '>', 18)->get();

// Multiple where clauses (AND)
$users = $db->where('status', 'active')
           ->where('role', 'admin')
           ->get();

// OR where
$users = $db->where('status', 'active')
           ->orWhere('status', 'pending')
           ->get();

// Where IN
$users = $db->whereIn('role', ['admin', 'editor', 'moderator'])->get();

// Where NULL
$users = $db->whereNull('deleted_at')->get();

// Where NOT NULL
$users = $db->whereNotNull('email_verified_at')->get();

Ordering and Limits

php
$db = new QueryBuilder('users');

// Order by
$users = $db->orderBy('name', 'ASC')->get();
$users = $db->orderBy('created_at', 'DESC')->get();

// Multiple order by
$users = $db->orderBy('status', 'ASC')
           ->orderBy('name', 'ASC')
           ->get();

// Limit
$users = $db->limit(10)->get();

// Offset (for pagination)
$users = $db->limit(10)->offset(20)->get();

Aggregates

php
$db = new QueryBuilder('users');

// Count
$total = $db->count();
$activeCount = $db->where('status', 'active')->count();

// Sum
$totalSalary = $db->sum('salary');

// Average
$averageAge = $db->avg('age');

// Maximum
$maxAge = $db->max('age');

// Minimum
$minAge = $db->min('age');

// Pluck (get single column values)
$names = $db->pluck('name'); // Returns ['John', 'Jane', ...]
$namesById = $db->pluck('name', 'id'); // Returns [1 => 'John', 2 => 'Jane']

Joins

php
$db = new QueryBuilder('orders');

// Inner join
$orders = $db->join('users', 'orders.user_id', '=', 'users.id')
             ->select('orders.*', 'users.name as user_name')
             ->get();

// Left join
$orders = $db->leftJoin('users', 'orders.user_id', '=', 'users.id')
             ->get();

// Right join
$orders = $db->rightJoin('users', 'orders.user_id', '=', 'users.id')
             ->get();

// Multiple joins
$orders = $db->join('users', 'orders.user_id', '=', 'users.id')
             ->join('products', 'orders.product_id', '=', 'products.id')
             ->get();

QueryBuilder with Entities (ORM Style)

Setting Entity Class

php

// Define your entity class
class User
{
    public $id;
    public $name;
    public $email;
    public $status;
    
    private $relations = [];
    
    public function __construct($data = [])
    {
        foreach ($data as $key => $value) {
            $this->$key = $value;
        }
    }
    
    public function getKey()
    {
        return $this->id;
    }
    
    public function setRelation($key, $value)
    {
        $this->relations[$key] = $value;
    }
    
    // Relationship methods
    public function posts()
    {
        return (new QueryBuilder('posts'))
            ->setEntity(Post::class)
            ->where('user_id', $this->id);
    }
    
    public function profile()
    {
        return (new QueryBuilder('profiles'))
            ->setEntity(Profile::class)
            ->where('user_id', $this->id);
    }
}

// Use QueryBuilder with entity
$users = (new QueryBuilder('users'))
    ->setEntity(User::class)
    ->where('status', 'active')
    ->get();

foreach ($users as $user) {
    echo $user->name . "\n"; // Access as object
}

Eager Loading Relationships

php
// Eager load relationships (prevents N+1 queries)
$users = (new QueryBuilder('users'))
    ->setEntity(User::class)
    ->with('posts', 'profile')
    ->where('status', 'active')
    ->get();

// With nested relationships
$users = (new QueryBuilder('users'))
    ->setEntity(User::class)
    ->with(['posts' => function($query) {
        $query->where('status', 'published')->orderBy('created_at', 'DESC');
    }])
    ->get();

// Load relationship counts
$users = (new QueryBuilder('users'))
    ->setEntity(User::class)
    ->withCount('posts', 'comments')
    ->get();

// Lazy load after query
$users = (new QueryBuilder('users'))->setEntity(User::class)->get();
$users = $users->load('posts'); // Load relationship

Advanced QueryBuilder Query Building

Complex Where Conditions

php
$db = new QueryBuilder('products');

// Combine multiple conditions
$products = $db->where('status', 'active')
               ->where('price', '>', 10)
               ->where('stock', '>', 0)
               ->orWhere('featured', 1)
               ->get();

// WHERE IN with array
$products = $db->whereIn('category_id', [1, 2, 3, 4])->get();

// WHERE NULL checks
$products = $db->whereNull('deleted_at')->get();
$products = $db->whereNotNull('published_at')->get();

Chunked Results (Memory Efficient)

php
$db = new QueryBuilder('users');

// Process large datasets in chunks
$page = 1;
$perPage = 100;

while (true) {
    $users = $db->limit($perPage)
                ->offset(($page - 1) * $perPage)
                ->get();
    
    if (empty($users)) {
        break;
    }
    
    foreach ($users as $user) {
        // Process user
        processUser($user);
    }
    
    $page++;
}

Subqueries

php
// Using multiple query builders
$posts = new QueryBuilder('posts');

// Get users with posts
$users = (new QueryBuilder('users'))
    ->whereIn('id', function($query) use ($posts) {
        return $posts->select('user_id')->get();
    })
    ->get();

User Repository with QueryBuilder

class UserRepository
{
    private $query;
    
    public function __construct()
    {
        $this->query = new QueryBuilder('users');
    }
    
    public function getAllActive($page = 1, $perPage = 20)
    {
        return $this->query->where('status', 'active')
                           ->orderBy('name', 'ASC')
                           ->paginate($perPage, $page);
    }
    
    public function findByEmail($email)
    {
        return $this->query->where('email', $email)->first();
    }
    
    public function createUser($data)
    {
        $data['password'] = password_hash($data['password'], PASSWORD_DEFAULT);
        $data['created_at'] = date('Y-m-d H:i:s');
        
        return $this->query->insert($data);
    }
    
    public function updateUser($id, $data)
    {
        return $this->query->where('id', $id)
                           ->setMultiple($data)
                           ->update();
    }
    
    public function deleteUser($id)
    {
        return $this->query->where('id', $id)->delete();
    }
    
    public function getUsersWithRecentOrders()
    {
        return $this->query->join('orders', 'users.id', '=', 'orders.user_id')
                           ->where('orders.created_at', '>', date('Y-m-d', strtotime('-30 days')))
                           ->select('users.*', 'orders.total', 'orders.created_at as order_date')
                           ->orderBy('orders.created_at', 'DESC')
                           ->get();
    }
    
    public function bulkUpdateStatus($userIds, $status)
    {
        $data = [];
        foreach ($userIds as $userId) {
            $data[] = ['id' => $userId, 'status' => $status];
        }
        
        return $this->query->updateBatch($data, 'id');
    }
}

Product Catalog with QueryBuilder

php
class ProductRepository
{
    private $query;
    
    public function __construct()
    {
        $this->query = new QueryBuilder('products');
    }
    
    public function search($filters)
    {
        $query = $this->query->where('status', 'active');
        
        if (!empty($filters['category'])) {
            $query->where('category_id', $filters['category']);
        }
        
        if (!empty($filters['min_price'])) {
            $query->where('price', '>=', $filters['min_price']);
        }
        
        if (!empty($filters['max_price'])) {
            $query->where('price', '<=', $filters['max_price']);
        }
        
        if (!empty($filters['search'])) {
            $query->where('name', 'LIKE', "%{$filters['search']}%")
                  ->orWhere('description', 'LIKE', "%{$filters['search']}%");
        }
        
        if (!empty($filters['sort'])) {
            switch ($filters['sort']) {
                case 'price_asc':
                    $query->orderBy('price', 'ASC');
                    break;
                case 'price_desc':
                    $query->orderBy('price', 'DESC');
                    break;
                case 'name_asc':
                    $query->orderBy('name', 'ASC');
                    break;
                default:
                    $query->orderBy('created_at', 'DESC');
            }
        }
        
        $perPage = $filters['per_page'] ?? 20;
        $page = $filters['page'] ?? 1;
        
        return $query->paginate($perPage, $page);
    }
    
    public function updateStock($productId, $quantity)
    {
        return $this->query->where('id', $productId)
                           ->set('stock', new RawExpression("stock - {$quantity}"))
                           ->update();
    }
    
    public function getLowStockProducts($threshold = 10)
    {
        return $this->query->where('stock', '<=', $threshold)
                           ->where('status', 'active')
                           ->orderBy('stock', 'ASC')
                           ->get();
    }
    
    public function bulkPriceUpdate($percentage, $operation = 'increase')
    {
        $operator = $operation === 'increase' ? '+' : '-';
        $expression = "price {$operator} (price * {$percentage} / 100)";
        
        return $this->query->set('price', new RawExpression($expression))
                           ->update();
    }
}

Order Processing with Transactions

php
class OrderService
{
    private $db;
    
    public function __construct()
    {
        $this->db = Database::getInstance();
    }
    
    public function createOrder($userId, $items)
    {
        try {
            $this->db->beginTransaction();
            
            // Calculate total
            $total = 0;
            foreach ($items as $item) {
                $product = (new QueryBuilder('products'))
                    ->where('id', $item['product_id'])
                    ->first();
                
                if (!$product || $product['stock'] < $item['quantity']) {
                    throw new \Exception("Insufficient stock for product ID: {$item['product_id']}");
                }
                
                $total += $product['price'] * $item['quantity'];
            }
            
            // Create order
            $orderId = (new QueryBuilder('orders'))->insert([
                'user_id' => $userId,
                'total' => $total,
                'status' => 'pending',
                'created_at' => date('Y-m-d H:i:s')
            ]);
            
            // Create order items and update stock
            foreach ($items as $item) {
                (new QueryBuilder('order_items'))->insert([
                    'order_id' => $orderId,
                    'product_id' => $item['product_id'],
                    'quantity' => $item['quantity']
                ]);
                
                (new QueryBuilder('products'))
                    ->where('id', $item['product_id'])
                    ->set('stock', new RawExpression("stock - {$item['quantity']}"))
                    ->update();
            }
            
            $this->db->commit();
            
            return $orderId;
            
        } catch (\Exception $e) {
            $this->db->rollBack();
            throw $e;
        }
    }
    
    public function getOrderWithDetails($orderId)
    {
        $order = (new QueryBuilder('orders'))
            ->where('id', $orderId)
            ->first();
        
        if ($order) {
            $order['items'] = (new QueryBuilder('order_items'))
                ->join('products', 'order_items.product_id', '=', 'products.id')
                ->select('order_items.*', 'products.name', 'products.price')
                ->where('order_items.order_id', $orderId)
                ->get();
            
            $order['user'] = (new QueryBuilder('users'))
                ->where('id', $order['user_id'])
                ->first();
        }
        
        return $order;
    }
}

QueryBuilder Best Practices

1. Always Use Parameter Binding

php
// GOOD - QueryBuilder automatically binds parameters
$users = $db->where('email', $userInput)->get();

// DON'T - Never concatenate strings
// $db->where("email = '{$userInput}'"); // SQL injection risk!

2. Use Entities for Complex Logic

php
// GOOD - Use entities when you have business logic
$users = (new QueryBuilder('users'))
    ->setEntity(User::class)
    ->with('posts')
    ->get();

// SIMPLE - Use array results for simple queries
$count = (new QueryBuilder('users'))->count();

3. Batch Process Large Datasets

php
// GOOD - Process in chunks
$page = 1;
while ($users = $db->limit(1000)->offset(($page-1)*1000)->get()) {
    foreach ($users as $user) {
        // Process
    }
    $page++;
}

4. Use RawExpression Carefully

php
// GOOD - Use RawExpression for database functions
$db->set('last_login', new RawExpression('NOW()'))->update();

// GOOD - Use for increment/decrement
$db->set('views', new RawExpression('views + 1'))->update();

5. Always Use Transactions for Related Operations

php
// GOOD - Transaction ensures data consistency
$db->beginTransaction();
try {
    $orderId = $orderQuery->insert($orderData);
    $itemQuery->insertBatch($items);
    $db->commit();
} catch (\Exception $e) {
    $db->rollBack();
    throw $e;
}

Getting Started with ORM

Basic Model Definition

php
// jephy-mvc/app/models/User.php
namespace App\Models;

use App\Core\Model;

class User extends Model
{
    protected static $table = 'users';
    
    // Define relationships (optional)
    public function posts()
    {
        return $this->hasMany(Post::class);
    }
    
    public function profile()
    {
        return $this->hasOne(Profile::class);
    }
    
    public function roles()
    {
        return $this->belongsToMany(Role::class, 'user_roles');
    }
}

class AuthController
{
    public function register($data)
    {
        // Check if email exists
        if (User::exists(['where' => ['email' => $data['email']]])) {
            return ['error' => 'Email already exists'];
        }
        
        // Create user
        $user = User::create([
            'name' => $data['name'],
            'email' => $data['email'],
            'password' => password_hash($data['password'], PASSWORD_DEFAULT),
            'status' => 'active',
            'created_at' => date('Y-m-d H:i:s')
        ]);
        
        return $user;
    }
    
    public function login($email, $password)
    {
        $user = User::findFirst([
            'where' => ['email' => $email, 'status' => 'active']
        ]);
        
        if ($user && password_verify($password, $user->password)) {
            $_SESSION['user_id'] = $user->id;
            return $user;
        }
        
        return null;
    }
    
    public function updateProfile($userId, $data)
    {
        $user = User::find($userId);
        
        if (!$user) {
            return null;
        }
        
        $user->update($data);
        return $user;
    }
}

    }

class ProductController
{
    public function search($filters)
    {
        $where = ['status' => 'active'];
        
        // Price range
        if (isset($filters['min_price'])) {
            $where['price'] = ['operator' => '>=', 'value' => $filters['min_price']];
        }
        
        if (isset($filters['max_price'])) {
            $where['price'] = ['operator' => '<=', 'value' => $filters['max_price']];
        }
        
        // Category filter
        if (isset($filters['category_id'])) {
            $where['category_id'] = $filters['category_id'];
        }
        
        // Search term
        if (isset($filters['search'])) {
            $where['name'] = ['contains' => $filters['search']];
        }
        
        $orderBy = $filters['sort'] ?? 'created_at DESC';
        $perPage = $filters['per_page'] ?? 20;
        $page = $filters['page'] ?? 1;
        $offset = ($page - 1) * $perPage;
        
        $products = Product::findMany([
            'where' => $where,
            'orderBy' => $orderBy,
            'take' => $perPage,
            'skip' => $offset
        ]);
        
        $total = Product::count(['where' => $where]);
        
        // Load category for each product
        foreach ($products as $product) {
            $product->load('category');
        }
        
        return [
            'products' => $products,
            'total' => $total,
            'page' => $page,
            'per_page' => $perPage,
            'total_pages' => ceil($total / $perPage)
        ];
    }
}
